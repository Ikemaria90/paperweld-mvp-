async function generateChecklist() { const state = document.getElementById('state').value; const treatment = document.getElementById('treatment').value || 'Aesthetics'; const pdfContent = document.getElementById('pdf-content'); const exportSection = document.getElementById('export-section'); const resultTitle = document.getElementById('result-title'); const tableBody = document.getElementById('result-table-body'); const reportMeta = document.getElementById('report-meta'); tableBody.innerHTML = `Generating AI compliance audit requirements for ${state}...`; pdfContent.style.display = 'block'; exportSection.style.display = 'block'; try { const res = await fetch('/api/generate', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ state, treatment }) }); const data = await res.json(); const requirements = data.requirements || []; const currentDate = new Date().toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' }); reportMeta.innerText = `Generated: ${currentDate} | Jurisdiction: ${state}`; resultTitle.innerText = `Compliance Audit Checklist: ${treatment} Practices in ${state}`; let rowsHTML = ''; requirements.forEach((req, index) => { let riskClass = 'risk-operational'; let riskText = req.risk || 'Operational'; if (riskText.toLowerCase().includes('critical')) { riskClass = 'risk-critical'; riskText = 'Critical'; } else if (riskText.toLowerCase().includes('best')) { riskClass = 'risk-best-practice'; riskText = 'Best Practice'; } rowsHTML += ` AI-${index + 1} ${req.item}: ${req.description} ${riskText}

`; }); tableBody.innerHTML = rowsHTML; // Programmatically bind click events to every status button to guarantee mobile responsiveness const statusButtons = tableBody.querySelectorAll('.status-action-btn'); statusButtons.forEach(button => { button.addEventListener('click', function() { if (this.classList.contains('pending')) { this.classList.remove('pending'); this.classList.add('compliant'); this.innerText = '✓ Compliant'; this.style.backgroundColor = '#238636'; } else { this.classList.remove('compliant'); this.classList.add('pending'); this.innerText = 'Pending'; this.style.backgroundColor = '#57606a'; } updateProgress(); }); }); updateProgress(); } catch (err) { tableBody.innerHTML = `Error communicating with AI service.`; } }
