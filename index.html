async function generateChecklist() { const state = document.getElementById('state').value; const treatment = document.getElementById('treatment').value || 'Aesthetics'; const pdfContent = document.getElementById('pdf-content'); const exportSection = document.getElementById('export-section'); const resultTitle = document.getElementById('result-title'); const tableBody = document.getElementById('result-table-body'); const reportMeta = document.getElementById('report-meta'); tableBody.innerHTML = `Generating AI compliance audit requirements for ${state}...`; pdfContent.style.display = 'block'; exportSection.style.display = 'block'; try { const res = await fetch('/api/generate', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ state, treatment }) }); const data = await res.json(); const requirements = data.requirements || []; const currentDate = new Date().toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' }); reportMeta.innerText = `Generated: ${currentDate} | Jurisdiction: ${state}`; resultTitle.innerText = `Compliance Audit Checklist: ${treatment} Practices in ${state}`; let rowsHTML = ''; requirements.forEach((req, index) => { rowsHTML += ` AI-${index + 1} ${req} Pending Review `; }); tableBody.innerHTML = rowsHTML; } catch (err) { tableBody.innerHTML = `Error communicating with AI service.`; } }
